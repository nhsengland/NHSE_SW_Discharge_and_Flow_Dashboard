    , cast([Date] AS date) AS 'Date'
    , [Org_Code], [Organisation_Name]
    , CASE WHEN [Metric_ID] = 'DIS005b_Total' 
                AND [Period] < '2024-05-27'

            WHEN [Metric_ID] = 'DIS006_TOTAL'

    , CASE  WHEN [STP_Name] = 'NHS BATH AND NORTH EAST SOMERSET, SWINDON AND WILTSHIRE INTEGRATED CARE BOARD' THEN 'Bath And North East Somerset, Swindon And Wiltshire STP'

            WHEN [STP_Name] = 'NHS CORNWALL AND THE ISLES OF SCILLY INTEGRATED CARE BOARD' THEN 'Cornwall And The Isles Of Scilly Health & Social Care Partnership (STP)'

            WHEN [STP_Name] = 'NHS SOMERSET INTEGRATED CARE BOARD' THEN 'Somerset STP'

    , sum(cast([Metric_Value] AS int)) AS 'Val'

            WHEN [STP] = 'NHS BRISTOL, NORTH SOMERSET AND SOUTH GLOUCESTERSHIRE INTEGRATED CARE BOARD' THEN 'Bristol, North Somerset And South Gloucestershire STP'
            WHEN [STP] = 'NHS CORNWALL AND THE ISLES OF SCILLY INTEGRATED CARE BOARD' THEN 'Cornwall And The Isles Of Scilly Health & Social Care Partnership (STP)'
            WHEN [STP] = 'NHS DEVON INTEGRATED CARE BOARD' THEN 'Devon STP'	
            WHEN [STP] = 'NHS DORSET INTEGRATED CARE BOARD' THEN 'Dorset STP'
            WHEN [STP] = 'NHS GLOUCESTERSHIRE INTEGRATED CARE BOARD' THEN 'Gloucestershire STP'
            WHEN [STP] = 'NHS SOMERSET INTEGRATED CARE BOARD' THEN 'Somerset STP'
                ELSE [STP] END AS [STP]
 , sum([Value]) AS 'Val'

FROM             [Reporting_DCF_EmergencyCare].[AnEDashboardMetricsTransposedDataTransport_Consolidated]

WHERE        Region_Name = 'South West' AND Period >= GETDATE() - 740 /*  and [Period] <= @_date*/ AND ([MetricName] IN ('Number_of_Admissions', 'AdultBedsClosedOtherUnoccupied', 'AdultTotalG&A_BedsOccupied', 
                         'AdultBedsClosedOther', 'AdultBedsClosedD&VUnoccupied', 'UnavailableBedStrandedPatients', 'G&ATotalBedsOccupied', 'G&ATotalBedsOpen', 'AdultEscalationG&A_BedsOpen', 'AdultCoreG&A_BedsOpen', 'G&AFluBeds',

                         'EmergencyAdmissionsA&E', 'AdultBedsClosedCovidUnoccupied'))
