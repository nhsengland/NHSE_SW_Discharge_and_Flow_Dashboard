WITH CTE AS (
 
SELECT

     CASE 

         WHEN Der_Provider_Code = 'RD3' THEN 'R0D'

         WHEN Der_Provider_Code = 'RDZ' THEN 'R0D'

         WHEN Der_Provider_Code = 'RBZ' THEN 'RH8'

         WHEN Der_Provider_Code = 'RA3' THEN 'RA7'

         WHEN Der_Provider_Code = 'RBA' THEN 'RH5'

         WHEN Der_Provider_Code = 'R1G' THEN 'R1G'

         WHEN Der_Provider_Code = 'RVJ13' THEN 'RVJ'

         WHEN Der_Provider_Code = 'RA4' THEN 'RH5'

         WHEN Der_Provider_Code = 'B4B4S' THEN 'RD1'

         ELSE Der_Provider_Code 

     END AS Provider_Code,

    CASE 

        WHEN o.organisation_name = 'ROYAL DEVON AND EXETER NHS FOUNDATION TRUST' 

            THEN 'ROYAL DEVON UNIVERSITY HEALTHCARE NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'TAUNTON AND SOMERSET NHS FOUNDATION TRUST' 

            THEN 'SOMERSET NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'NORTHERN DEVON HEALTHCARE NHS TRUST' 

            THEN 'ROYAL DEVON UNIVERSITY HEALTHCARE NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'THE ROYAL BOURNEMOUTH AND CHRISTCHURCH HOSPITALS NHS FOUNDATION TRUST' 

            THEN 'UNIVERSITY HOSPITAL DORSET NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'POOLE HOSPITAL NHS FOUNDATION TRUST' 

            THEN 'UNIVERSITY HOSPITAL DORSET NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'TORBAY AND SOUTHERN DEVON HEALTH AND CARE NHS TRUST' 

            THEN 'TORBAY AND SOUTH DEVON NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'EMERSONS GREEN NHS TREATMENT CENTRE' 

            THEN 'NORTH BRISTOL NHS TRUST'

        WHEN o.organisation_name = 'YEOVIL DISTRICT HOSPITAL NHS FOUNDATION TRUST' 

            THEN 'SOMERSET NHS FOUNDATION TRUST'	 

        WHEN o.Organisation_Name = 'UNIVERSITY HOSPITALS BRISTOL NHS FOUNDATION TRUST' 

            THEN 'UNIVERSITY HOSPITALS BRISTOL AND WESTON NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'BSW BANES LOCALITY CDC' 

            THEN 'ROYAL UNITED HOSPITALS BATH NHS FOUNDATION TRUST'

        ELSE o.organisation_name  

    END AS Org_Name,
 
    FORMAT(apc.Discharge_Date, '01/MM/yyyy') AS Discharge_Date,

    CONCAT(YEAR(apc.Discharge_Date), FORMAT(MONTH(apc.Discharge_Date), '00')) AS YearMonth,
 
    o.STP_Name,

    o.STP_Code,
 
    apc.Der_Management_Type AS Management_Type,
 
    CASE

        WHEN DATEPART(HOUR, apc.discharge_time) >= 12 THEN 'Post-Noon'

        WHEN DATEPART(HOUR, apc.discharge_time) < 12 THEN 'Pre-Noon'

        ELSE apc.discharge_time

    END AS Discharge_Hour,
 
    CASE

        WHEN DATEPART(weekday, apc.Discharge_Date) IN (1, 7) THEN 'Weekend'

        ELSE 'Weekday'

    END AS Weekend_Weekday,
 
    COUNT(CAST(apc.[APCS_Ident] AS VARCHAR)) AS Patients,
 
    CONCAT(YEAR(apc.Discharge_Date), FORMAT(MONTH(apc.Discharge_Date), '00')) AS YearMonth_Sort
 
FROM [MESH_APC].[APCS_Core_Daily] apc
 
LEFT JOIN [Internal_Reference].[UCSM_RefTFCs] AS tfc

    ON apc.Der_Dischg_Treatment_Function_Code = tfc.Treatment_Function_Code
 
LEFT JOIN [Reporting_UKHD_ODS].[Provider_Hierarchies] o

    ON apc.Der_Provider_Code = o.Organisation_Code COLLATE Latin1_General_CI_AS
 
LEFT JOIN [MESH_APC].[APCS_2526_Der_Daily] AS c

    ON apc.APCS_Ident = c.[APCS_Ident]
 
WHERE

    apc.Discharge_Date >= DATEADD(
        MONTH,
        -24,
        DATEFROMPARTS(YEAR(GETDATE()), MONTH(GETDATE()), 1)
    )

    AND apc.Discharge_Date < DATEFROMPARTS(
        YEAR(GETDATE()),
        MONTH(GETDATE()),
        1
    )

  


 
    AND c.Responsible_Purchaser_Assignment_Method NOT IN ('Private Patient')

    AND apc.Der_Spell_Los > 0
 
    AND apc.Der_Dischg_Treatment_Function_Code NOT IN (

        '199','223','290','291','331','344','345','346','360','424','499','501','504','560',

        '650','651','652','653','654','655','656','657','658','659','660','661','662',

        '700','710','711','712','713','715','720','721','722','723','724','725','726',

        '727','730','840','920','NULL'

    )
 
    AND Der_Management_Type IN ('NE','EM','EL')
 
    AND Der_Provider_Code IN (

        'RD1','RN3','RNZ','RA7','RVJ','RH8','RK9',

        'RBD','R0D','RTE','RH5','RVJ13','B4B4S','R1G'

    )
 
    AND Der_Provider_Code <> 'REF'
 
    -- 🚫 EXCLUDE RA9 COMPLETELY

    AND Der_Provider_Code <> 'RA9'
 
    AND o.STP_Code <> 'QT6'
 
    AND NOT (

        apc.Der_Provider_Code = 'RTE'

        AND apc.Discharge_Date >= '2023-04-01'

        AND apc.Discharge_Date <= '2024-03-31'

    )
 
GROUP BY

    CASE 

         WHEN Der_Provider_Code = 'RD3' THEN 'R0D'

         WHEN Der_Provider_Code = 'RDZ' THEN 'R0D'

         WHEN Der_Provider_Code = 'RBZ' THEN 'RH8'

         WHEN Der_Provider_Code = 'RA3' THEN 'RA7'

         WHEN Der_Provider_Code = 'RBA' THEN 'RH5'

         WHEN Der_Provider_Code = 'R1G' THEN 'R1G'

         WHEN Der_Provider_Code = 'RVJ13' THEN 'RVJ'

         WHEN Der_Provider_Code = 'RA4' THEN 'RH5'

         WHEN Der_Provider_Code = 'B4B4S' THEN 'RD1'

         ELSE Der_Provider_Code 

    END,
 
    CASE 

        WHEN o.organisation_name = 'ROYAL DEVON AND EXETER NHS FOUNDATION TRUST' 

            THEN 'ROYAL DEVON UNIVERSITY HEALTHCARE NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'TAUNTON AND SOMERSET NHS FOUNDATION TRUST' 

            THEN 'SOMERSET NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'NORTHERN DEVON HEALTHCARE NHS TRUST' 

            THEN 'ROYAL DEVON UNIVERSITY HEALTHCARE NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'THE ROYAL BOURNEMOUTH AND CHRISTCHURCH HOSPITALS NHS FOUNDATION TRUST' 

            THEN 'UNIVERSITY HOSPITAL DORSET NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'POOLE HOSPITAL NHS FOUNDATION TRUST' 

            THEN 'UNIVERSITY HOSPITAL DORSET NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'TORBAY AND SOUTHERN DEVON HEALTH AND CARE NHS TRUST' 

            THEN 'TORBAY AND SOUTH DEVON NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'EMERSONS GREEN NHS TREATMENT CENTRE' 

            THEN 'NORTH BRISTOL NHS TRUST'

        WHEN o.organisation_name = 'YEOVIL DISTRICT HOSPITAL NHS FOUNDATION TRUST' 

            THEN 'SOMERSET NHS FOUNDATION TRUST'	 

        WHEN o.Organisation_Name = 'UNIVERSITY HOSPITALS BRISTOL NHS FOUNDATION TRUST' 

            THEN 'UNIVERSITY HOSPITALS BRISTOL AND WESTON NHS FOUNDATION TRUST'

        WHEN o.organisation_name = 'BSW BANES LOCALITY CDC' 

            THEN 'ROYAL UNITED HOSPITALS BATH NHS FOUNDATION TRUST'

        ELSE o.organisation_name 

    END,
 
    FORMAT(apc.Discharge_Date, '01/MM/yyyy'),

    CONCAT(YEAR(apc.Discharge_Date), FORMAT(MONTH(apc.Discharge_Date), '00')),

    o.STP_Name,

    o.STP_Code,

    apc.Der_Management_Type,
 
    CASE

        WHEN DATEPART(HOUR, apc.discharge_time) >= 12 THEN 'Post-Noon'

        WHEN DATEPART(HOUR, apc.discharge_time) < 12 THEN 'Pre-Noon'

        ELSE apc.discharge_time

    END,
 
    CASE

        WHEN DATEPART(weekday, apc.Discharge_Date) IN (1, 7) THEN 'Weekend'

        ELSE 'Weekday'

    END,
 
    CONCAT(YEAR(apc.Discharge_Date), FORMAT(MONTH(apc.Discharge_Date), '00'))

)
 
SELECT *

FROM CTE
 
UNION ALL
 
SELECT

    'South West Total' AS Provider_Code,

    'South West Total' AS Org_Name,

    Discharge_Date,

    YearMonth,

    'South West Total' AS STP_Name,

    'South West Total' AS STP_Code,

    Management_Type,

    Discharge_Hour,

    Weekend_Weekday,

    Patients,

    YearMonth_Sort

FROM CTE
