SELECT
    a.Discharge_Ready_Date,
    a.Discharge_Date,
    DATEFROMPARTS(YEAR(a.Discharge_Date), MONTH(a.Discharge_Date), 1) AS Month_Start, -- Month column
    a.Der_Provider_Code AS Orgcode,
    LEFT(a.Discharge_Time, 2) AS [Discharge hour],
    a.Discharge_Destination,
    a.Der_Management_Type,
    a.Der_Provider_Site_Code AS site_code,
    b.STP_Name,
    COUNT(CAST(a.APCE_Ident AS varchar)) AS MetricValue,
    
    -- Age Band column
    CASE
        WHEN a.Age_at_End_of_Episode_SUS BETWEEN 0 AND 0 THEN '0'
        WHEN a.Age_at_End_of_Episode_SUS BETWEEN 1 AND 18 THEN '1-18'
        WHEN a.Age_at_End_of_Episode_SUS BETWEEN 19 AND 39 THEN '19-39'
        WHEN a.Age_at_End_of_Episode_SUS BETWEEN 40 AND 64 THEN '40-64'
        WHEN a.Age_at_End_of_Episode_SUS >= 65 THEN '65+'
        ELSE 'Missing/Invalid'
    END AS Age_Band,

    -- Days Group column based on Discharge_Date - Discharge_Ready_Date
    CASE
        WHEN a.Discharge_Date IS NULL OR a.Discharge_Ready_Date IS NULL THEN 'null'
        WHEN DATEDIFF(day, a.Discharge_Ready_Date, a.Discharge_Date) = 0 THEN '0'
        WHEN DATEDIFF(day, a.Discharge_Ready_Date, a.Discharge_Date) <= 7 THEN '1-7'
        WHEN DATEDIFF(day, a.Discharge_Ready_Date, a.Discharge_Date) <= 14 THEN '8-14'
        WHEN DATEDIFF(day, a.Discharge_Ready_Date, a.Discharge_Date) <= 20 THEN '14-20'
        WHEN DATEDIFF(day, a.Discharge_Ready_Date, a.Discharge_Date) >= 21 THEN '21+'
        ELSE 'Missing/Invalid'
    END AS Days_Group

FROM [MESH_APC].[APCE_Core_Daily] AS a
LEFT OUTER JOIN [Reporting_UKHD_ODS].[Provider_Hierarchies_ICB] AS b 
    ON a.Der_Provider_Code = b.Organisation_Code

WHERE
    a.Hospital_Spell_Duration > '0'
    AND a.Admission_Method NOT IN ('31', '32', '82', '83')
    AND a.Discharge_Destination NOT IN ('51', '52', '88', '87', '79')
    AND a.Age_at_End_of_Episode_SUS >= '16'
    AND a.Treatment_Function_Code NOT IN ('360', '812')
    AND a.Episode_End_Date >= GETDATE() - 120
    AND a.Last_Episode_In_Spell_Indicator = '1'
    AND a.Der_Provider_Code IN ('RbZ', 'RVJ', 'RH5', 'RA3', 'R0D', 'RA9', 'RH8', 
                                'RA7', 'RBD', 'RNZ', 'RA4', 'RN3', 'RK9', 'RTE', 'RD1', 'REF')

GROUP BY
    a.Der_Provider_Code,
    a.Der_Provider_Site_Code,
    b.STP_Name,
    a.Discharge_Ready_Date,
    a.Discharge_Destination,
    a.Der_Management_Type,
    a.Discharge_Date,
    LEFT(a.Discharge_Time, 2),
    a.Age_at_End_of_Episode_SUS
