DECLARE @MonthStart DATE =
    DATEFROMPARTS(
        YEAR(GETDATE()),
        MONTH(GETDATE()),
        1
    );
    
    
    
    /* ===============================
   ORG LEVEL – DISCHARGE
   =============================== */
SELECT 
    'Discharge' AS [Table],
    CAST([Date] AS DATE) AS [Date],

    CASE
        WHEN DATENAME(WEEKDAY, CAST([Date] AS DATE)) IN ('Saturday','Sunday')
            THEN 'Weekend'
        ELSE 'Weekday'
    END AS Weekend_Weekday,

    [Org_Code],
    [Organisation_Name],

    CASE [Org_Code]
        WHEN 'RN3' THEN 'QOX'
        WHEN 'RD1' THEN 'QOX'
        WHEN 'RNZ' THEN 'QOX'
        WHEN 'RVJ' THEN 'QUY'
        WHEN 'RA7' THEN 'QUY'
        WHEN 'REF' THEN 'QT6'
        WHEN 'RH8' THEN 'QJK'
        WHEN 'RA9' THEN 'QJK'
        WHEN 'RK9' THEN 'QJK'
        WHEN 'RBD' THEN 'QVV'
        WHEN 'R0D' THEN 'QVV'
        WHEN 'RTE' THEN 'QR1'
        WHEN 'RH5' THEN 'QSL'
        ELSE NULL
    END AS STP_Code,

    CASE 
        WHEN [Metric_ID] = 'DIS005b_Total' AND [Period] < '2024-05-27' THEN '21d'
        WHEN [Metric_ID] = 'DIS006_TOTAL' AND [Period] >= '2024-05-27' THEN '21d'
        WHEN [Metric_ID] = 'DIS003_Total' THEN 'DIS004a_Total'
        WHEN [Metric_ID] = 'DIS003b_TOTAL' THEN 'DIS004b_Total'
        ELSE Metric_ID 
    END AS Met_id_adj,

    CASE 
        WHEN [Metric_ID] = 'DIS002_Total' THEN 'The number of people who do not meet the criteria to reside that day'
        WHEN [Metric_ID] = 'DIS003_TOTAL' THEN 'Of the total number of people who do not meet the criteria to reside that day: the number of people who were discharged by 17.00'
        WHEN [Metric_ID] = 'DIS003b_TOTAL' THEN 'Of the total number of people who do not meet the criteria to reside that day: the number of people who were discharged between 17.01 and 23.59'
        ELSE [Metric_Name] 
    END AS Metric_Name,

    [STP_Name],
    SUM(CAST([Metric_Value] AS INT)) AS Val

FROM [Reporting].[SEFT_CovidSitreps_Discharge_DashboardMetrics]

WHERE Region = 'South West'
  AND CAST([Period] AS DATE) >= DATEADD(MONTH, -24, @MonthStart)
AND CAST([Period] AS DATE) < @MonthStart
  AND Metric_ID IN (
        'DIS002_Total','DIS003_Total','DIS004a_Total',
        'DIS003b_Total','DIS004b_Total','DIS005b_Total','DIS006_Total'
  )

GROUP BY
    CAST([Date] AS DATE),
    CASE
        WHEN DATENAME(WEEKDAY, CAST([Date] AS DATE)) IN ('Saturday','Sunday')
            THEN 'Weekend'
        ELSE 'Weekday'
    END,
    [Org_Code],
    [Organisation_Name],
    [STP_Name],
    [Metric_ID],
    [Metric_Name],
    [Period]

UNION ALL

/* ===============================
   ORG LEVEL – AE
   =============================== */

    'AE' AS [Table],
    CAST([Period] AS DATE) AS [Date],

    CASE
        WHEN DATENAME(WEEKDAY, CAST([Period] AS DATE)) IN ('Saturday','Sunday')
            THEN 'Weekend'
        ELSE 'Weekday'
    END AS Weekend_Weekday,

    [OrgCode] AS Org_Code,
    [Trust_Name] AS Organisation_Name,

    CASE [OrgCode]
        WHEN 'RN3' THEN 'QOX'
        WHEN 'RD1' THEN 'QOX'
        WHEN 'RNZ' THEN 'QOX'
        WHEN 'RVJ' THEN 'QUY'
        WHEN 'RA7' THEN 'QUY'
        WHEN 'REF' THEN 'QT6'
        WHEN 'RH8' THEN 'QJK'
        WHEN 'RA9' THEN 'QJK'
        WHEN 'RK9' THEN 'QJK'
        WHEN 'RBD' THEN 'QVV'
        WHEN 'R0D' THEN 'QVV'
        WHEN 'RTE' THEN 'QR1'
        WHEN 'RH5' THEN 'QSL'
        ELSE NULL
    END AS STP_Code,

    [MetricName] AS Met_id_adj,
    [MetricName] AS Metric_Name,
    [STP] AS STP_Name,

    SUM([Value]) AS Val

FROM [Reporting_DCF_EmergencyCare].[AnEDashboardMetricsTransposedDataTransport_Consolidated]

WHERE Region_Name = 'South West'
  AND CAST([Period] AS DATE) >= DATEADD(MONTH, -24, @MonthStart)
AND CAST([Period] AS DATE) < @MonthStart
  AND MetricName IN (
        'Number_of_Admissions','AdultBedsClosedOtherUnoccupied',
        'AdultTotalG&A_BedsOccupied','AdultBedsClosedOther',
        'AdultBedsClosedD&VUnoccupied','UnavailableBedStrandedPatients',
        'G&ATotalBedsOccupied','G&ATotalBedsOpen',
        'AdultEscalationG&A_BedsOpen','AdultCoreG&A_BedsOpen',
        'G&AFluBeds','AdultTotalG&A_BedsOpen',
        'Beds_occupied_by_long_stay_patients_14_plus_days',
        'AdultBedsClosedD&V','Extended_Patients_Occupied',
        'AdultBedsClosedCovid','EmergencyAdmissions',
        'EmergencyAdmissionsA&E','AdultBedsClosedCovidUnoccupied'
  )

GROUP BY
    CAST([Period] AS DATE),
    CASE
        WHEN DATENAME(WEEKDAY, CAST([Period] AS DATE)) IN ('Saturday','Sunday')
            THEN 'Weekend'
        ELSE 'Weekday'
    END,
    [OrgCode],
    [Trust_Name],
    [STP],
    [MetricName]

UNION ALL
